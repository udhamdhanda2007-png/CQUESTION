Q1

Write a program to print 'A' using recursion.

ANSWER
#include <stdio.h>

void printA(int n)
{
    if (n == 0)
        return;
    printf("A ");
    printA(n - 1);
}

void main()
{
    printA(5);
}
Output
A A A A A
Q2

Write a program to print numbers from 1 to 5 using recursion.

ANSWER
#include <stdio.h>

void print(int n)
{
    if (n > 5)
        return;
    printf("%d ", n);
    print(n + 1);
}

void main()
{
    print(1);
}
Output
1 2 3 4 5
Q3

ANSWER

void print(int n)

    print(5);
}

Q5

Write a program to find the factorial of a number using recursion.

ANSWER
#include <stdio.h>

int fact(int n)
{
    if (n == 0 || n == 1)
        return 1;
    return n * fact(n - 1);
}

void main()
{
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);
    printf("Factorial of %d = %d\n", num, fact(num));
}
Output
Enter a number: 5
Factorial of 5 = 120
