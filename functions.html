Q4

Make the following programs using local and global variable: Factorial, reverse, palindrome, prime number, armstrong, fibonacci?

ANSWER
#include<stdio.h>
#include<conio.h>

int num;   //global variable
void factorial();
void reverse();
void palindrome();
void prime();
void armstrong();
void fibonacci();

void main()
{
   clrscr();
   printf("Enter a number:");
   scanf("%d",&num);
   factorial();
   reverse();
   palindrome();
   prime();
   armstrong();
   fibonacci();
   getch();
}

void factorial()
{
   int i,fact=1; //local variables
   for(i=1;i<=num;i++)
   {
      fact *=i;
   }
   printf("\nfactorial of %d=%d",num,fact);
}
void reverse()
{
   int n=num,rev=0;
   while(n>0)
   {
      rev=rev*10+n%10;
      n/=10;
   }
   printf("\nreverse of %d=%d",num,rev);
}
void palindrome()
{
   int n=num,rev=0,temp;
   temp=n;
   while(n>0)
   {
      rev=rev*10+n%10;
      n/=10;
   }
   if(temp==rev)
      printf("\n%d is a palindrome number",num);
   else
      printf("\n%d is not a palindrome number",num);
}
void prime()
{
   int i, count=0;
   for(i=1;i<=num;i++)
   {
      if(num%i==0)
         count++;
   }
   if(count==2)
      printf("\n%d is a prime number", num);
   else
      printf("\n%d is not a prime number",num);
}
void armstrong()
{
   int n=num,rem,sum=0;
   while(n>0)
   {
      rem=n%10;
      sum+=rem*rem*rem;
      n/=10;
   }
   if(sum==num)
      printf("\n%d is an armstrong number",num);
   else
      printf("\n%d is not an armstrong number",num);
}
void fibonacci()
{
   int n=num;
   int a=0,b=1,c,i;
   printf("\nFibonacci series up to %d terms:",n);
   for(i=1;i<=n;i++)
   {
      printf("%d ",a);
      c=a+b;
      a=b;
      b=c;
   }
}
Output

Q5

Swap numbers using call by address?

ANSWER
#include <stdio.h>
#include<conio.h>

// Function to swap numbers using call by address
void swap(int *a, int *b) {
    int temp;
    temp = *a;   // Store value of a in temp
    *a = *b;     // Put value of b into a
    *b = temp;   // Put temp (old a) into b
}

void main() {
    int num1, num2;
    clrscr();

    printf("Enter two numbers: ");
    scanf("%d %d", &num1, &num2);

    printf("\nBefore swapping:\n");
    printf("num1 = %d, num2 = %d\n", num1, num2);

    swap(&num1, &num2);

    printf("\nAfter swapping:\n");
    printf("num1 = %d, num2 = %d\n", num1, num2);
}
Output
Enter two numbers: 10 20

Before swapping:
num1 = 10, num2 = 20

After swapping:
num1 = 20, num2 = 10
