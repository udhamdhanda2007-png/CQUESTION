ANSWER

Q4

Making following program using local and global variable: Factorial, Reverse, Palindrome, Prime numbers, Armstrong, Fibonacci.

ANSWER
#include <stdio.h>
int n;
void input() {
    printf("Enter number: ");
    scanf("%d", &n);
}
void factorial() {
    int i,f=1;
    for(i=1;i<=n;i++)
        f=f*i;
    printf("Factorial = %d\n", f);
}
void reverse() {
    int x, i=0, y;
    printf("enter a number: ");
    scanf("%d", &x);
    while(x > 0){
        y = x % 10;
        i = i * 10 + y;
        x = x / 10;
    }
    printf("Reverse = %d\n", i);
}
void palindrome() {
    int n, t, r, s = 0;
    printf("enter no: ");
    scanf("%d", &n);
    t = n;
    while(n > 0){
        r = n % 10;
        s = s * 10 + r;
        n = n / 10;
    }
    if(t == s)
        printf("Palindrome = Yes\n");
    else
        printf("Palindrome = No\n");
}
void prime() {
    int i, c = 0;
    for(i = 1; i <= n; i++) {
        if(n % i == 0)
            c++;
    }
    if(c == 2)
        printf("Prime = Yes\n");
    else
        printf("Prime = No\n");
}
void armstrong() {
    int temp = n, sum = 0, r;
    while(temp != 0) {
        r = temp % 10;
        sum += r * r * r;
        temp /= 10;
    }
    if(sum == n)
        printf("Armstrong = Yes\n");
    else
        printf("Armstrong = No\n");
}
void fibonacci() {
    int a = 0, b = 1, c;
    printf("Fibonacci: %d %d ", a, b);
    for(int i = 2; i <= n; i++) {
        c = a + b;
        printf("%d ", c);
        a = b;
        b = c;
    }
    printf("\n");
}
void main() {
    input();
    factorial();
    reverse();
    palindrome();
    prime();
    armstrong();
    fibonacci();
}

Q5

Swap numbers using call by address.

ANSWER
#include <stdio.h>
void swap(int *x, int *y){
    int z;
    z=*x;
    *x=*y;
    *y=z;
}
void og(int a,int b){
    printf("original value of \na=%d\nb=%d\n", a, b);
}
void main(){
    int a,b;
    a = 20;
    b = 10;
    og(a, b);
    swap(&a, &b);
    printf("swaped values are \n");
    printf("a= %d\n", a);
    printf("b= %d\n", b);
}

a=20
b=10
swaped values are 
a= 10
b= 20

Q8

Make a function to find out the cube of a user entered number.

ANSWER
#include <stdio.h>
int input(){
    int x;
    printf("enter number: ");
    scanf("%d", &x);
    return x;
}
void cube(int a){
    int z;

}

    a = input();
    cube(a);
}
