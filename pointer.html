Q1

Write a program to enter 5 values in an array using pointer and print them.

ANSWER
#include <stdio.h>

void main()
{
    int arr[5];
    int *p;
    int i;

    p = arr;

    for (i = 0; i < 5; i++)
    {
        printf("Enter value for location %d: ", i);
        scanf("%d", p + i);
    }

    printf("\nArray elements are:\n");
    for (i = 0; i < 5; i++)
    {
        printf("arr[%d] = %d\n", i, *(p + i));
    }
}
Output
Enter value for location 0: 10
Enter value for location 1: 20
Enter value for location 2: 30
Enter value for location 3: 40
Enter value for location 4: 50

Array elements are:
arr[0] = 10
arr[1] = 20
arr[2] = 30
arr[3] = 40
arr[4] = 50
Q2

Write a program to enter values in a 2D array using a pointer and print the matrix.

ANSWER
#include <stdio.h>

void main()
{
    int arr[2][3];
    int *p;
    int i, j;

    p = &arr[0][0];

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 3; j++)
        {
            printf("Enter value for arr[%d][%d]: ", i, j);
            scanf("%d", p + (i * 3 + j));
        }
    }

    printf("\nThe 2D array is:\n");
    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 3; j++)
        {
            printf("%d\t", *(p + (i * 3 + j)));
        }
        printf("\n");
    }
}
Output
Enter value for arr[0][0]: 1
Enter value for arr[0][1]: 2
Enter value for arr[0][2]: 3
Enter value for arr[1][0]: 4
Enter value for arr[1][1]: 5
Enter value for arr[1][2]: 6

The 2D array is:
1   2   3
4   5   6
