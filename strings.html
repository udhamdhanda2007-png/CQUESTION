Q1

Enter a string from user and convert in uppercase.

ANSWER
#include <stdio.h>
#include <Strings.h>

void main(){
    char str[50];
    int i;
    printf("Enter any sentence you like: ");
    gets(str);
    for(i=0;str[i];i++){
        if(str[i]>='a'&& str[i]<='z'){
            str[i]=str[i]-32;
        }
    }
    printf("Here is the uppercase version of it: %s",str);
}
Output
Enter any sentence you like: Hello World
Here is the uppercase version of it: HELLO WORLD

Q18

String palindrome example (using reverse).

ANSWER
#include <stdio.h>
#include <strings.h>
void main(){
    char str1[50],str2[50];
    int i,l;
    printf("Pallindrome are words that spells the same forward and backward for example \nNITIN,NAMAN and MOM\n");
    printf("Enter any word to check if it is a pallindrome: ");
    gets(str1);
    l=strlen(str1);
    for(i=0;str1[i];i++){
        str2[i]=str1[l-1-i];
    }
    str2[i]=0;
    l=strcmp(str1,str2);
    if(l==0)printf("That is a pallindrome");
    else printf("That is not a pallindrome");
}
Output
Pallindrome are words that spells the same forward and backward for example
NITIN,NAMAN and MOM
Enter any word to check if it is a pallindrome: sus
That is a pallindrome
