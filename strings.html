Q4

Enter a string from the user and convert it into title case.

ANSWER

#include<conio.h>
#include<ctype.h>

{
   char str[50];
   int i=0;
   clrscr();

   while(str[i]!='\0')
   {
      str[i]=tolower(str[i]);
      i++;
   }
   if(str[0]>='a'&&str[0]<='z')
   {
      str[0]=toupper(str[0]);
   }

   i=0;
   while(str[i]!='\0')
   {
      if(str[i]==' '&&str[i+1]>='a'&&str[i+1]<='z')
      {
         str[i+1]=toupper(str[i+1]);
      }
      i++;
   }
   printf("String in title case: %s\n",str);
}
Output

String in title case: Hello World Of C

Q18

String palindrome example (using reverse).

ANSWER
#include<stdio.h>
#include<conio.h>
#include<string.h>

void main()
{
   char str[50],rev[50];
   int i,len;
   clrscr();
   printf("Enter a string: ");
   gets(str);
   len=strlen(str);
   for(i=0;i<len;i++)
   {
      rev[i]=str[len-1-i];
   }
   rev[len]='\0';
   if(strcmp(str,rev)==0)
      printf("%s is a palindrome\n",str);
   else
      printf("%s is not a palindrome\n",str);
}
Output
Enter a string: madam
madam is a palindrome
