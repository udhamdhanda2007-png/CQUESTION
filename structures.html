Q1

Create structure of employee which has emp.ID, name, designation of 50 employees from user, print details in tabular form.

ANSWER
#include<stdio.h>
#include<conio.h>

struct employee
{
   int id;
   char name[50];
   char designation[50];
};
void main()
{
   struct employee emp[50];
   int i,n;
   clrscr();
   printf("Enter number of employees(max 50): ");
   scanf("%d",&n);
   if(n>50)
   {
      printf("You can only enter up to 50 emplyees.\n");
      return;
   }
   for(i=0;i<n;i++)
   {
      printf("Enter details of emplo yees %d: \n",i+1);
      printf("Enter employee id: ");
      scanf("%d",&emp[i].id);
      printf("Enter employee name: ");
      scanf("%s",emp[i].name);
      printf("Enter employee designation: ");
      scanf("%s",emp[i].designation);
   }
   printf("\n---------------------------------------------------------\n");
   printf("%-10s %-20s %-20s\n","emp.ID","Name","Designation");
   printf("-----------------------------------------------------------\n");
   for(i=0;i<n;i++)
   {
      printf("%-10d %-20s %-20s\n",emp[i].id,emp[i].name,emp[i].designation);
   }
   printf("------------------------------------------------------------\n");
}

Q3

Create structure of employee which has emp.ID, name, designation of 50 employees from user, print name character by character with emp.ID in tabular form.

ANSWER
#include<stdio.h>
#include<conio.h>
#include<string.h>

struct employee
{
   int id;
   char name[50];
   char designation[50];
};
void main()
{
   struct employee emp[50];
   int i,j,n;
   clrscr();
   printf("Enter number of employess(max 50): ");
   scanf("%d",&n);
   getchar();
   for(i=0;i<n;i++)
   {
      printf("Enter details of employee %d:\n",i+1);
      printf("Enter employee ID: ");
      scanf("%d",&emp[i].id);
      getchar();
      printf("Enter employee name: ");
      gets(emp[i].name);
      printf("Enter employee designation: ");
      gets(emp[i].designation);
   }
   printf("\nEmp.Id\tName(character by character)\n");
   printf("------------------------------------------------\n");
   for(i=0;i<n;i++)
   {
      printf("%d\t",emp[i].id);
      for(j=0;emp[i].name[j]!='\0';j++)
      {
         printf("%c",emp[i].name[j]);
      }
      printf("\n");
   }
}

Q4

Enter student details like roll, name, fee, DOB and print all the details in ascending order of roll number.

ANSWER
#include<stdio.h>
#include<conio.h>
#include<string.h>

struct student
{
   int roll;
   char name[50];
   int fee;
   char dob[15];
};
void main()
{
   struct student s[100],temp;
   int n,i,j;
   clrscr();
   printf("Enter number of students: ");
   scanf("%d",&n);

   for(i=0;i<n;i++)
   {
      printf("\nEnter details of student %d:\n",i+1) ;
      printf("Roll Number: ");
      scanf("%d",&s[i].roll);
      printf("Name: ");
      scanf("%s",s[i].name);
      printf("Fee: ");
      scanf("%d",&s[i].fee);
      printf("Date of birth(DD/MM/YYYY): ");
      scanf("%s",s[i].dob);
   }

   for(i=0;i<n;i++)
   {
      for(j=i+1;j<n;j++)
      {
         if(s[i].roll>s[j].roll)
         {
            temp=s[i];
            s[i]=s[j];
            s[j]=temp;
         }
      }
   }
   printf("\n Students details in Ascending order of roll number: \n");
   printf("-------------------------------------------------------------------\n");
   printf("Roll\tName\t\tFee\t\tDOB\n");
   printf("------------------------------------------------------------------\n");
   for(i=0;i<n;i++)
   {
      printf("%d\t%-10s\t%.2d\t\t%s\n",s[i].roll,s[i].name,s[i].fee,s[i].dob);
   }
}

Q4

Enter student details like roll number, name, fee, and DOB and print them in ascending order of roll number.

ANSWER
#include <stdio.h>
struct student
{
    int roll;
    char name[20];
    int fee;
    char dob[20];
};

int main()
{
    struct student stu[5], temp;

    printf("Enter details of 5 students:\n");

    for (int i = 0; i < 5; i++)
    {

        printf("\nStudent %d:\n", i + 1);
        printf("Enterc your roll number: ");
        scanf("%d", &stu[i].roll);
        fflush(stdin);

        printf("Enter your Name: ");
        scanf("%s", stu[i].name);
        fflush(stdin);

        printf("Enter the fees: ");
        scanf("%d", stu[i].fee);
        fflush(stdin);

        printf("Enter the DOB: ");
        scanf("%s", stu[i].dob);
        fflush(stdin);
    }

    for (int i = 0; i < 5 - 1; i++)
    {
        for (int j = i + 1; j < 5; j++)
        {
            if (stu[i].roll > stu[j].roll)
            {
                temp = stu[i];
                stu[i] = stu[j];
                stu[j] = temp;
            }
        }
    }

    printf("\n-------------------------------------------------------------\n");
    printf("| %-10s | %-15s | %-10s | %-12s |\n", "Roll No", "Name", "Fee", "DOB");
    printf("-------------------------------------------------------------\n");

    for (int i = 0; i < 5; i++)
    {
        printf("| %-10d | %-15s | %-10d | %-12s |\n",
               stu[i].roll, stu[i].name, stu[i].fee, stu[i].dob);
    }

    printf("-------------------------------------------------------------\n");

    return 0;
}
