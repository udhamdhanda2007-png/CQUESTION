Q22

Enter current date and DOB of user and find out difference?

ANSWER
#include <stdio.h>
#include<conio.h>

void main() {
    int c_day, c_month, c_year;
    int b_day, b_month, b_year;
    int age_day, age_month, age_year;

    printf("Enter current date (dd mm yyyy): ");
    scanf("%d %d %d", &c_day, &c_month, &c_year);

    printf("Enter date of birth (dd mm yyyy): ");
    scanf("%d %d %d", &b_day, &b_month, &b_year);

    if (c_day < b_day) {
        c_day += 30;
        c_month -= 1;
    }

    if (c_month < b_month) {
        c_month += 12;
        c_year -= 1;
    }

    age_day = c_day - b_day;
    age_month = c_month - b_month;
    age_year = c_year - b_year;

    printf("\nYour Age is: %d years, %d months, and %d days.\n",
           age_year, age_month, age_day);
}
Output
Enter current date (dd mm yyyy): 3 11 2026
Enter date of birth (dd mm yyyy): 15 8 2005

Your Age is: 21 years, 2 months, and 18 days.
