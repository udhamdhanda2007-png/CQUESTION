Q1

Write a program to show an example of pre-increment.

ANSWER
#include <stdio.h>
void main() {
    int a = 5;
    int b = ++a;
    printf("a=%d, b=%d", a, b);
}
Output
a=6, b=6

Q5

Write a program to enter 2 numbers from user, add, subtract, multiply, divide and find the remainder, and display the answers.

ANSWER
#include <stdio.h>
void main() {
    int a,b;
    printf("Enter two numbers for addition: ");
    scanf("%d %d", &a, &b);
    printf("Addition=%d\n", a+b);

    printf("Enter two numbers for subtraction: ");
    scanf("%d %d", &a, &b);
    printf("Subtraction=%d\n", a-b);

    printf("Enter two numbers for multiplication: ");
    scanf("%d %d", &a, &b);
    printf("Multiplication=%d\n", a*b);

    printf("Enter two numbers for division: ");
    scanf("%d %d", &a, &b);
    printf("Division=%d\n", a/b);

    printf("Enter two numbers for remainder: ");
    scanf("%d %d", &a, &b);
    printf("Remainder=%d\n", a%b);
}
Output
Enter two numbers for addition: 2 2
Addition=4
Enter two numbers for subtraction: 2 2
Subtraction=0
Enter two numbers for multiplication: 2 2
Multiplication=4
Enter two numbers for division: 2 2
Division=1
Enter two numbers for remainder: 2 2
Remainder=0

Q11

Enter marks of students and calculate average of the student’s marks.

ANSWER
#include <stdio.h>
void main() {
    int n, i, marks, sum=0;
    printf("Enter number of students: ");
    scanf("%d", &n);
    for(i=1;i<=n;i++){
        printf("Enter marks of student %d: ", i);
        scanf("%d", &marks);
        sum += marks;
    }
    printf("Average marks: %.2f", (float)sum/n);
}
Output
Enter number of students: 3
Enter marks of student 1: 77
Enter marks of student 2: 58
Enter marks of student 3: 98
Average marks: 77.67
Q12

Calculate simple interest = (P x R x T) / 100.

ANSWER
#include <stdio.h>
void main() {
    float P, R, T, SI;
    printf("Enter Principal, Rate, Time: ");
    scanf("%f %f %f", &P, &R, &T);
    SI = (P*R*T)/100;
    printf("Simple Interest: %.2f", SI);
}
Output
Enter Principal, Rate, Time: 2 3 4
Simple Interest: 0.24
