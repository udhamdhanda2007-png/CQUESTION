Q1

Write a program to show example of pre-increment.

ANSWER

#include<conio.h>

{

    clrscr();

Output
THE VALUE OF X: 6
THE VALUE OF Y: 6

Q5

Enter two numbers from the user and add them and show the answer; again take 2 numbers, subtract them; again multiply; again divide; again find the remainder and show the answer.

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
    int num1,num2,sum;
    clrscr();
    printf("Enter first number:");
    scanf("%d",&num1);
    printf("Enter second number:");
    scanf("%d",&num2);
    sum=num1+num2;   //ADDITION.
    printf("ADDITION %d\n",sum);

    printf("Enter first number:");
    scanf("%d",&num1);
    printf("Enter second number:");
    scanf("%d",&num2);
    sum=num1-num2;   //SUBTRACTION.
    printf("SUBTRACT %d\n",sum);

    printf("Enter first number:");
    scanf("%d",&num1);
    printf("Enter second number:");
    scanf("%d",&num2);
    sum=num1*num2;     //MULTIPLICATION.
    printf("MULTILPLE %d\n",sum);

    printf("Enter first number:");
    scanf("%d",&num1);
    printf("Enter second number:");
    scanf("%d",&num2);
    sum=num1/num2;      //DIVISION.
    printf("DIVISION %d\n",sum);

    printf("Enter first number:");
    scanf("%d",&num1);
    printf("Enter second number:");
    scanf("%d",&num2);
    sum=num1%num2;       //REMAINDER
    printf("REMAINDER %d\n",sum);
}
Output

Enter second number:5

Enter first number:10

Enter first number:10

Q7

Enter a decimal number from the user (float) and print the right hand side of the decimal point?

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
    float num;
    int intpart,rightside;
    float fracpart;
    clrscr();
    printf("Enter a Decimal number:");
    scanf("%f",&num);

    intpart=(int)num;
    fracpart=num-intpart;
    rightside=(int)(fracpart*1000000);
    while(rightside%10==0&&rightside!=0)
       rightside/=10;
    printf("Right-hand side of decimal point :%d\n",rightside);
}
Output
Enter a Decimal number:12.75
Right-hand side of decimal point :75

Q11

Enter marks of students and calculate average of students marks?

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
    int marks1,marks2,marks3,marks4,marks5;
    float avg;
    clrscr();
    printf("enter the marks of first subject:-");
    scanf("%d",&marks1);
    printf("enter the marks of second subject:-");
    scanf("%d",&marks2);
    printf("enter the marks of third subject:-");
    scanf("%d",&marks3);
    printf("enter the marks of fourth subject:-");
    scanf("%d",&marks4);
    printf("enter the marks of fifth subject:-");
    scanf("%d",&marks5);

enter the marks of second subject:-70
enter the marks of third subject:-90
enter the marks of fourth subject:-60
