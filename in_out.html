Q6

Write a program for printing your name on output screen but your name should blink five times.

ANSWER

#include<conio.h>
#include<dos.h>

void main()
{
    int i;
    clrscr();
    for(i=0;i<5;i++)
    {
        printf("UDHAM");

        delay(500);
    }
}

Q12

Enter password, display like ******.

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
    char password[50];
    int i=0;
    char ch;
    clrscr();
    printf("enter password: ");
    while(1)
    {
        ch=getch();
        if(ch==13)
        {
            password[i]='\0';
            break;
        }
        else if(ch==8)
        {
            if(i>0)
            {
                i--;
                printf("\b \b");
            }
        }
        else
        {
            password[i++]=ch;
            printf("*");
        }
    }
    printf("\nyour password is:%s\n",password);
}
Output
enter password: ***
your password is:abc

Q14

Print all the ASCII characters 0 to 127.

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
    int i;
    printf("ASCII table (0 to 127)\n");
    for(i=0;i<128;i++)
    {
        if(i<32||i==127)
            printf("%3d\t non-printable\n",i);
        else
            printf("%3d\t%c\n",i,i);
    }
}
Output

  0	 non-printable
  1	 non-printable
  ...
