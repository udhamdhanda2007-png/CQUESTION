Q6

Write a Turbo C program to blink your name 5 times.

ANSWER
#include <stdio.h>
#include <conio.h>
#include <dos.h>

void main()
{
    int count;
    clrscr();
    for (count = 0; count < 5; count++) {
        gotoxy(35, 12);
        textcolor(BLINK);
        cprintf("Your Name");
        delay(500);
        gotoxy(35, 12);
        cprintf("         ");
        delay(500);
    }
    textcolor(WHITE);
    getch();
}
Output
Your Name blinks on the screen 5 times.

Q9

Write a Turbo C program to enter a password and display one asterisk for each character.

ANSWER
#include <stdio.h>
#include <conio.h>

void main()
{
    char password[21];
    int length = 0;
    char ch;

    printf("Enter password (max 20 characters): ");
    while ((ch = getch()) != '\r' && length < 20) {
        if (ch == '\b' && length > 0) {
            length--;
            printf("\b \b");
        } else if (ch != '\b') {
            password[length++] = ch;
            printf("*");
        }
    }
    password[length] = '\0';
    printf("\nPassword entered.\n");
}
Output
Enter password (max 20 characters): ******
Password entered.
Q10

Write a program to enter two characters and print their ASCII values.

ANSWER
#include <stdio.h>

void main()
{
    char first, second;
    printf("Enter two characters: ");
    scanf(" %c %c", &first, &second);
    printf("ASCII value of %c = %d\n", first, (unsigned char) first);
    printf("ASCII value of %c = %d\n", second, (unsigned char) second);
}
Output
Enter two characters: A B
ASCII value of A = 65
ASCII value of B = 66
Q11

Write a program to print all ASCII characters from 0 to 127. Control characters are shown by name.

ANSWER
#include <stdio.h>

void main()
{
    const char *controlNames[] = {
        "NUL", "SOH", "STX", "ETX", "EOT", "ENQ", "ACK", "BEL",
        "BS", "TAB", "LF", "VT", "FF", "CR", "SO", "SI",
        "DLE", "DC1", "DC2", "DC3", "DC4", "NAK", "SYN", "ETB",
        "CAN", "EM", "SUB", "ESC", "FS", "GS", "RS", "US"
    };
    int code;

    for (code = 0; code <= 127; code++) {
        printf("%3d: ", code);
        if (code < 32) {
            printf("%s", controlNames[code]);
        } else if (code == 127) {
            printf("DEL");
        } else {
            printf("%c", code);
        }
        printf("\n");
    }
}
Output
  0: NUL
  1: SOH
  2: STX
  ...
 31: US
 32: (space)
 33: !
 65: A
 97: a
126: ~
127: DEL
(The program prints every code from 0 through 127.)
