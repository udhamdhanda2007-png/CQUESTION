#include<conio.h>

{

   for(i=0;i<5;i++)
   {
      scanf("%d",&arr[i]);

   printf("Enter value to search: ");

      if(arr[i]==value)

         found=1;
         break;

   }

      printf("Found\n");

      printf("not found\n");
}

Q11

Enter 5 values of array and sort in ascending order.

ANSWER
#include<stdio.h>
#include<conio.h>

void main()
{
   int arr[5],i,j,temp;
   clrscr();
   printf("Enter 5 elements: \n");
   for(i=0;i<5;i++)
   {
      scanf("%d",&arr[i]);
   }
   for(i=0;i<5;i++)
   {
      for (j=i+1;j<5;j++)
      {
         if(arr[i]>arr[j])
         {
            temp=arr[i];
            arr[i]=arr[j];
            arr[j]=temp;
         }
      }
   }
   printf("Array elements in ascending order: \n");
   for(i=0;i<5;i++)
   {
      printf("%d \n",arr[i]);
   }
}
Output
Enter 5 elements:
50
10
40
20
30
Array elements in ascending order:
10
20
30
40
50
